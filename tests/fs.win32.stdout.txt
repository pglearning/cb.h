== glob 匹配（纯函数）==

cb_glob_match("a*c", "abbbc") = 1

cb_glob_match("a*c", "ab") = 0
cb_glob_match("?.c", "a.c") = 1
cb_glob_match("?.c", "ab.c") = 0（? 只匹配一个字符）
cb_glob_match("[abc]x", "bx") = 1
cb_glob_match("[abc]x", "dx") = 0
cb_glob_match("[a-z]1", "q1") = 1
cb_glob_match("[a-z]1", "Q1") = 0
cb_glob_match("[!abc]x", "dx") = 1
cb_glob_match("[!abc]x", "ax") = 0
cb_glob_match("src/*/*.c", "src/a/b.c") = 1
cb_glob_match("a*b*c", "aXXbYYc") = 1
cb_glob_match("a*b*c", "aXXbYY") = 0
cb_glob_match("", "") = 1
cb_glob_match("", "x") = 0
