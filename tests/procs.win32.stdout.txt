
== cb_fd_open_write / cb_fd_open_read / cb_fd_close ==
CB_FD fd = CB_INVALID_FD; fd == CB_INVALID_FD -> 1
cb_fd_open_write("fd.txt") != CB_INVALID_FD -> 1
Windows：CB_FD 是 HANDLE，字节读写要用 WriteFile/ReadFile，这里只验证打开/关闭
cb_fd_open_read("fd.txt") != CB_INVALID_FD -> 1
cb_fd_open_read("no-such-file.txt") == CB_INVALID_FD -> 1

== cb_pipe_create / CB_Pipe ==
cb_pipe_create(&pp) -> 1
pp.read 与 pp.write 都不等于 CB_INVALID_FD -> 1
Windows：管道读写要用 WriteFile/ReadFile，跳过字节往返
cb_fd_close(pp.read) 与 cb_fd_close(pp.write) 已调用（fd 数值不打印）

== cb_proc_wait / cb__proc_wait_async / CB_INVALID_PROC ==
CB_Proc invalid = CB_INVALID_PROC; invalid == CB_INVALID_PROC -> 1
cb_proc_wait(CB_INVALID_PROC) -> 0（无效进程直接返回 false）
cb__proc_wait_async(CB_INVALID_PROC, 0) -> 0（无效进程直接返回 0）
cb__cmd_start_process(cmd, NULL, NULL, NULL) != CB_INVALID_PROC -> 1
轮询 cb__proc_wait_async(proc, 0) 直到返回 1 -> 1
cb__cmd_start_process 起第二个进程 != CB_INVALID_PROC -> 1
cb_proc_wait(proc2) 阻塞到进程退出并返回退出码 0 -> 1

== cb_cmd_run(.async, .max_procs) / cb_procs_wait / CB_Procs ==
cb_cmd_run(.async = &procs, .max_procs = 1) #1 -> 1，procs.count = 1
cb_cmd_run(.async = &procs, .max_procs = 1) #2 -> 1，procs.count = 1
cb_cmd_run(.async = &procs, .max_procs = 1) #3 -> 1，procs.count = 1
cb_procs_wait(procs) -> 1（收尾剩下的 1 个）
cb_procs_wait 之后 procs.count = 1（只等待，不重置数组）
cb_cmd_run(.async = &many, .max_procs = 4) #1 -> 1，many.count = 1
cb_cmd_run(.async = &many, .max_procs = 4) #2 -> 1，many.count = 2
cb_cmd_run(.async = &many, .max_procs = 4) #3 -> 1，many.count = 3
cb_procs_wait(many) -> 1
cb_nprocs()（可并行度，按 CPU 核数）>= 1 -> 1
